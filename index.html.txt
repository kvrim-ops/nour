    <script>
        function checkPass() {
            const pass = document.getElementById('passInput').value;
            if(pass === "Koki&Noni") {
                document.getElementById('step1').classList.add('hidden');
                document.getElementById('step2').classList.remove('hidden');
            } else {
                alert("كلمة السر غلط يا نونتي! ركزي 😂");
            }
        }

        function checkDate() {
            const date = document.getElementById('dateInput').value;
            if(date === "11/01/2026") {
                document.getElementById('step2').classList.add('hidden');
                document.getElementById('step3').classList.remove('hidden');
                startTimer();
            } else {
                alert("التاريخ غلط! افتكري اليوم اللي حياتي نورت فيه ❤️");
            }
        }

        function playMusic() {
            const music = document.getElementById('bgMusic');
            music.play();
            document.getElementById('playBtn').innerText = "شغالة الآن.. 🎵";
        }

        function startTimer() {
            const startDate = new Date("2026-01-11T00:00:00").getTime();
            setInterval(function() {
                const now = new Date().getTime();
                const diff = now - startDate;
                const days = Math.floor(diff / (1000 * 60 * 60 * 24));
                const hours = Math.floor((diff % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
                const mins = Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60));
                const secs = Math.floor((diff % (1000 * 60)) / 1000);
                document.getElementById('timer').innerHTML = `بقالنا مع بعض حبايب: <br> ${days} يوم و ${hours} ساعة و ${mins} دقيقة و ${secs} ثانية`;
            }, 1000);
        }
    </script>
